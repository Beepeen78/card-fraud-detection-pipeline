// DAX Measures for Fraud Detection Dashboard
// Copy these into Power BI Desktop → Modeling → New Measure

// ============================================
// BASIC METRICS
// ============================================

Total Transactions = COUNTROWS(Transactions)

Total Fraud Detected = SUM(Transactions[fraud_prediction])

Fraud Rate = DIVIDE([Total Fraud Detected], [Total Transactions], 0)

Average Fraud Probability = AVERAGE(Transactions[fraud_probability])

Total Amount at Risk = 
CALCULATE(
    SUM(Transactions[amt]),
    Transactions[fraud_prediction] = 1
)

Average Fraud Amount = 
CALCULATE(
    AVERAGE(Transactions[amt]),
    Transactions[fraud_prediction] = 1
)

// ============================================
// RISK LEVEL BREAKDOWN
// ============================================

Low Risk Count = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[RiskLevel] = "Low"
)

Medium Risk Count = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[RiskLevel] = "Medium"
)

High Risk Count = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[RiskLevel] = "High"
)

Critical Risk Count = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[RiskLevel] = "Critical"
)

// ============================================
// TIME-BASED METRICS
// ============================================

Fraud This Hour = 
CALCULATE(
    [Total Fraud Detected],
    FILTER(
        Transactions,
        Transactions[Hour] = HOUR(NOW())
    )
)

Fraud Today = 
CALCULATE(
    [Total Fraud Detected],
    FILTER(
        Transactions,
        Transactions[TransactionDate] >= TODAY()
    )
)

Fraud This Week = 
CALCULATE(
    [Total Fraud Detected],
    FILTER(
        Transactions,
        Transactions[TransactionDate] >= TODAY() - 7
    )
)

Fraud This Month = 
CALCULATE(
    [Total Fraud Detected],
    FILTER(
        Transactions,
        Transactions[TransactionDate] >= EOMONTH(TODAY(), -1) + 1
    )
)

// ============================================
// AMOUNT-BASED METRICS
// ============================================

Average Transaction Amount = AVERAGE(Transactions[amt])

Total Transaction Amount = SUM(Transactions[amt])

High Value Fraud Count = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[fraud_prediction] = 1,
    Transactions[amt] > 500
)

// ============================================
// PERFORMANCE METRICS (if ground truth available)
// ============================================

True Positives = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[fraud_prediction] = 1,
    Transactions[is_fraud] = 1
)

False Positives = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[fraud_prediction] = 1,
    Transactions[is_fraud] = 0
)

False Negatives = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[fraud_prediction] = 0,
    Transactions[is_fraud] = 1
)

True Negatives = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[fraud_prediction] = 0,
    Transactions[is_fraud] = 0
)

Precision = 
DIVIDE(
    [True Positives],
    [True Positives] + [False Positives],
    0
)

Recall = 
DIVIDE(
    [True Positives],
    [True Positives] + [False Negatives],
    0
)

F1 Score = 
DIVIDE(
    2 * [Precision] * [Recall],
    [Precision] + [Recall],
    0
)

Accuracy = 
DIVIDE(
    [True Positives] + [True Negatives],
    [Total Transactions],
    0
)

// ============================================
// TREND METRICS
// ============================================

Fraud Rate Trend = 
VAR CurrentRate = [Fraud Rate]
VAR PreviousRate = 
    CALCULATE(
        [Fraud Rate],
        DATEADD(Transactions[TransactionDate], -1, DAY)
    )
RETURN
    CurrentRate - PreviousRate

Fraud Count Change = 
VAR CurrentCount = [Total Fraud Detected]
VAR PreviousCount = 
    CALCULATE(
        [Total Fraud Detected],
        DATEADD(Transactions[TransactionDate], -1, DAY)
    )
RETURN
    CurrentCount - PreviousCount

// ============================================
// PERCENTAGE METRICS
// ============================================

Low Risk Percentage = 
DIVIDE([Low Risk Count], [Total Transactions], 0)

Medium Risk Percentage = 
DIVIDE([Medium Risk Count], [Total Transactions], 0)

High Risk Percentage = 
DIVIDE([High Risk Count], [Total Transactions], 0)

Critical Risk Percentage = 
DIVIDE([Critical Risk Count], [Total Transactions], 0)
